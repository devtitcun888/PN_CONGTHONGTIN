                   PageSize="@pageSize"
                   CurrentPage="@currentPage"
                   OnPageChanged="@OnPageChanged"
                   OnPageSizeChanged="@OnPageSizeChanged" />

<Pagination_Custom TotalItems="@total"
                   PageSize="@pageSize"
                   CurrentPage="@currentPage"
                   OnPageChanged="@OnPageChanged"
                   ShowPageSizeSelector="false"
                   PageSizeOptions="@(new[] { 5, 10, 20 })" />

private int currentPage = 1;
private int pageSize    = 15;

private void OnPageChanged(int page)
{
    currentPage = page;
    BuildPagedList();  // Skip/Take theo currentPage & pageSize
    StateHasChanged();
}

private void OnPageSizeChanged(int size)
{
    pageSize    = size;
    currentPage = 1;
    BuildPagedList();
    StateHasChanged();
}


Parameter	Kiểu	Default	Mô tả
TotalItems	int	—	Bắt buộc — tổng số bản ghi
PageSize	int	15	Số dòng mỗi trang
CurrentPage	int	1	Trang hiện tại (1-based)
OnPageChanged	EventCallback<int>	—	Callback khi chuyển trang
OnPageSizeChanged	EventCallback<int>	—	Callback khi đổi số dòng/trang
ShowPageSizeSelector	bool	true	Ẩn/hiện dropdown chọn pageSize
ShowInfo	bool	true	Ẩn/hiện dòng "Hiển thị X–Y / Z"
PageSizeOptions	int[]	[10,15,25,50]	Tùy chỉnh danh sách chọn
